export function ramp({ h, s, l }, steps = 9) {
  return Array.from({ length: steps }, (_, i) => ({
    step: (i + 1) * 100,
    color: `hsl(${h} ${s}% ${l + (4 - i) * 8}%)`,
  }));
}
